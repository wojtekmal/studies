\begin{zadanie}{440}
Które z następujących stwierdzeń są prawdziwe dla dowolnego porządku $(X, \le)$ i dowolnych $A, B \subseteq X$? A jeśli ten porządek jest liniowy?
\begin{enumerate}[label=(\alph*)]
    \item Jeśli istnieje $\sup(A \cup B)$, to istnieją $\sup A$ i $\sup B$?
    \item Jeśli istnieją $\sup A$ i $\sup B$, to istnieje $\sup(A \cup B)$?
    \item Jeśli istnieją $\sup A$, $\sup B$ i $\sup(A \cup B)$, to $\sup(A \cup B) = \sup\{\sup A, \sup B\}$?
\end{enumerate}
\end{zadanie}
\begin{rozwiazanie}
\begin{enumerate}[label=(\alph*)]
    \item Niech $A = [0, 1), B = (2, 3]$ oraz $X = A \cup B$ i $\leq$ to relacja mniejsze lub równe na liczbach rzeczywistych, o której wiemy, że jest liniową relacją częściowego porządku. 3 jest jedynym ograniczeniem górnym, więc $\sup(A\cup B) = 3$ istnieje. Jeśli $x \in X$ jest ograniczeniem górnym, to $x \in B$, ale wtedy $\frac{x+2}{2} < x$ też jest ograniczeniem górnym, czyli $x$ nie jest supremum $A$. Zatem $\sup A$ nie istnieje.
    \item Bez wymagania na liniowość możemy wziąć taki kontrprzykład: $X = \{a, b\}, A = \{a\}, B = \{b\}$ oraz każdy element jest w relacji tylko z samym sobą. Wtedy $\sup A = a$, $\sup B = b$, ale $\sup(A \cup B) = \sup\{a, b\}$ nie istnieje, bo nie ma ograniczeń górnych. Jeśli relacja jest liniowa, to niech $a = \sup A, b = \sup B$. Skoro relacja jest liniowa, to jedno z $a, b$ jest większe od drugiego; BSO niech to będzie $a$. Wtedy $\forall_{x\in A} a \geq x$ i $\forall_{x\in B} a \geq b \geq x$, czyli $\forall_{x\in A\cup B} a \geq x$. Jeśli jest jakieś inne ograniczenie górne $c$ na $A \cup B$, to jest to ograniczenie górne na $A$, czyli $c \geq a$. Zatem $a = \sup(A\cup B)$.
    \item 
\end{enumerate}
\end{rozwiazanie}

\begin{zadanie}{456}
Niech $r$ będzie częściowym porządkiem w zbiorze $A$ i niech $f: A \xrightarrow[na]{1-1} A$. Rozpatrzmy funkcję $g: A \times A \to A \times A$, określoną tak: $g(x,y) = \langle f(x), f(y) \rangle$. Udowodnić, że:
\begin{enumerate}[label=(\alph*)]
    \item $g(r) \subseteq r$ wtedy i tylko wtedy, gdy $f$ jest monotoniczna ze względu na porządek $r$.
    \item $r \subseteq g(r)$ wtedy i tylko wtedy, gdy $\forall x,y \, (f(x) \, r \, f(y) \to x \, r \, y)$.
\end{enumerate}
\end{zadanie}
\begin{rozwiazanie}
\begin{enumerate}[label=(\alph*)]
    \item $f$ jest monotoniczna w.t.w. gdy dla dowolnych $x, y \in A$, $x r y$ implikuje $f(x) r f(y)$. Każdy element $g(r)$ jest postaci $(f(x), f(y))$, gdzie $xry$, więc to w.t.w. gdy każda taka para $(f(x), f(y)) \in r$, czyli to w.t.w. gdy $g(r) \subseteq r$.
    \item Skoro $f$ jest bijekcją, to każde $(a, b) \in r$ da się jednoznacznie przedstawić jako $(f(x), f(y))$.
\end{enumerate}
\end{rozwiazanie}

\begin{zadanie}{622}
Jakiej mocy jest zbiór wszystkich łańcuchów
\begin{itemize}[label=--]
    \item w zbiorze $\mathbb{N} - \{0\}$, uporządkowanym przez relację podzielności?
    \item w zbiorze słów nad alfabetem $\{a, b\}$, uporządkowanym prefiksowo?
    \item w zbiorze słów nad alfabetem $\{a, b\}$, uporządkowanym leksykograficznie?
\end{itemize}
\end{zadanie}
\begin{rozwiazanie}
\begin{itemize}[label=--]
    \item Łańcuchy są zbiorami liczb naturalnych, a zbiorów liczb naturalnych jest $|\mathsf{P}(\N)| = \mathfrak{c}$, więc jest co najwyżej $\mathfrak{c}$ łańcuchów. Niech $X$ to zbiór łańcuchów. Mamy funkcję $\varphi : \P^\N \to \mathsf{P}(\N)$ zdefiniowaną tak:
    \[\varphi(f) = \{\prod_{k=1}^{n} f(k): n \in \N\}\]
    Wynika to wprost z definicji, że elementy $\varphi(f)$ się dzielą, czyli to jest łańcuch. Jeśli $n$ to pierwsza pozycja na której $f(n) \neq g(n)$, to
    \[\prod_{k=1}^{n} f(k) \neq \prod_{k=1}^{n} g(k)\]
    To są jedyne elementy $\varphi(f)$ i $\varphi(g)$, które mają $k$ dzielników pierwszych, czyli skoro są różne, to $\varphi(f) \neq \varphi(g)$, zatem $\varphi \in \P^\N \to X$ jest injekcją. Zatem $|X| \geq |\P^\N| = \aleph_0^{\aleph_0} = \mathfrak{c}$. Zatem $|X| = \mathfrak{c}$.
    \item Niech $\varphi : \{a,b\}^\N \to \powerset(\{a, b\}^*)$ jest takie, że przypisuje każdemu nieskończonemu ciągowi zbiór jego prefiksów. Prefiks do $i$-tej pozycji jest prefiksem prefiksu do $j$-tej pozycji, czyli $\varphi(f)$ jest łańcuchem. Podobnie jak w poprzednim podpunkcie, jeśli dwa nieskończone ciągi liter różnią się na $n$-tej pozycji, to prefiksy długości $n$ będą różne, czyli $\varphi$ jest injekcją. Zatem $\varphi \in \{a,b\}^\N \to^{1-1} X$, gdzie $X$ to zbiór łańcuchów, czyli $|X| \geq |\{a,b\}^\N| = |\{0,1\}^\N| = \mathfrak{c}$. $X \subseteq \powerset(\{a, b\}^*)$, a słów nad $\{a, b\}$ jest tyle, co skończonych ciągów binarnych, czyli tyle, co liczb naturalnych, które są reprezentowane przez te skończone ciągi, czyli $|X| \leq |\powerset(\{a, b\}^*)| = 2^{\aleph_0} = \mathfrak{c}$, czyli $|X| = \mathfrak{c}$.
    \item Każdy zbiór będący łańcuchem w porządku prefiksowym jest łańcuchem w porządku leksykograficznym, czyli przynajmniej $\mathfrak{c}$. Są to też podzbiory $\powerset(\{a, b\}^*)$, czyli jest ich co najwyżej $\mathfrak{c}$. Zatem odpowiedź to $\mathfrak{c}$.
\end{itemize}
\end{rozwiazanie}
